function computeQuality(data,cols,numCols){
  const n=Math.max(1,data.length),cells=Math.max(1,n*cols.length),filled=cols.reduce((s,c)=>s+data.filter(r=>!isMissing(r[c])).length,0),completeness=100*filled/cells;
  let validChecks=0,validOk=0,consistencyPen=0,integrity=100,timeliness=70,accuracy=100;
  for(const c of cols){const vals=data.map(r=>r[c]).filter(v=>!isMissing(v));if(state.types[c]==="numérique"){validChecks+=vals.length;validOk+=vals.filter(v=>smartNumber(v)!==null).length}if(state.types[c]==="date"){validChecks+=vals.length;const ds=vals.map(smartDate).filter(Boolean);validOk+=ds.length;if(ds.length){const maxd=new Date(Math.max(...ds));const age=(Date.now()-maxd.getTime())/86400000;timeliness=Math.max(20,Math.min(100,100-age/18))}}if(["catégorielle","texte"].includes(state.types[c])){const norms=new Map();for(const v of vals){const k=normalizeText(v);norms.set(k,(norms.get(k)||new Set()).add(String(v).trim()))}consistencyPen+=[...norms.values()].reduce((s,x)=>s+Math.max(0,x.size-1),0)}}
  const validity=validChecks?100*validOk/validChecks:90;
  const uniqueness=100*(1-state.duplicates/n);
  const id=state.business?.primaryId;if(id){const vals=data.map(r=>r[id]);const valid=vals.filter(v=>!isMissing(v)),uniq=new Set(valid.map(String)).size;integrity=.55*(100*valid.length/n)+.45*(100*uniq/Math.max(1,valid.length))}else integrity=55;
  const consistency=Math.max(0,100-100*consistencyPen/cells*8);
  const anomalyRate=(state.anomalyDetails?.length||state.anomalies.length)/n;accuracy=Math.max(0,100-anomalyRate*160);
  const dims={completeness:clamp(completeness,0,100),validity:clamp(validity,0,100),consistency:clamp(consistency,0,100),uniqueness:clamp(uniqueness,0,100),integrity:clamp(integrity,0,100),timeliness:clamp(timeliness,0,100),accuracy:clamp(accuracy,0,100)};
  const score=.20*dims.completeness+.16*dims.validity+.14*dims.consistency+.14*dims.uniqueness+.14*dims.integrity+.10*dims.timeliness+.12*dims.accuracy;
  state.qualityDimensions=dims;state.qualityIssues=collectQualityIssues(data,cols,state.types,state.stats);
  return{score:clamp(score,0,100),completeness:dims.completeness,uniqueness:dims.uniqueness,consistency:dims.consistency,anomalyHealth:dims.accuracy,...dims};
}
function computeAnomalies(data,numCols,stats){
  const details=[],summaries=new Map(),id=state.business?.primaryId;

function betacf(a,b,x){const MAX=120,EPS=3e-10,FPMIN=1e-30;let qab=a+b,qap=a+1,qam=a-1,c=1,d=1-qab*x/qap;if(Math.abs(d)<FPMIN)d=FPMIN;d=1/d;let h=d;for(let m=1;m<=MAX;m++){let m2=2*m,aa=m*(b-m)*x/((qam+m2)*(a+m2));d=1+aa*d;if(Math.abs(d)<FPMIN)d=FPMIN;c=1+aa/c;if(Math.abs(c)<FPMIN)c=FPMIN;d=1/d;h*=d*c;aa=-(a+m)*(qab+m)*x/((a+m2)*(qap+m2));d=1+aa*d;if(Math.abs(d)<FPMIN)d=FPMIN;c=1+aa/c;if(Math.abs(c)<FPMIN)c=FPMIN;d=1/d;const del=d*c;h*=del;if(Math.abs(del-1)<EPS)break}return h}
